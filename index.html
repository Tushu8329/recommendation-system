# GenAI Practical 1: Generative AI vs Discriminative Models
import numpy as np
import matplotlib.pyplot as plt
from tensorflow.keras.datasets import mnist
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense, Flatten, Reshape

# --------------------------------------------------
# 1. Load MNIST Dataset
# --------------------------------------------------

(x_train, y_train), (x_test, y_test) = mnist.load_data()

# Normalize pixel values: 0-255 -> 0-1
x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0

print("Training images:", x_train.shape)
print("Testing images :", x_test.shape)


# --------------------------------------------------
# 2. Discriminative Model - Classification
# --------------------------------------------------

classifier = Sequential([
    Flatten(input_shape=(28, 28)),
    Dense(128, activation="relu"),
    Dense(10, activation="softmax")
])

classifier.compile(
    optimizer="adam",
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)

print("\nTraining Discriminative Model...")

classifier.fit(
    x_train,
    y_train,
    epochs=3,
    batch_size=128,
    validation_split=0.1
)

loss, accuracy = classifier.evaluate(x_test, y_test, verbose=0)

print("\nDiscriminative Model Accuracy:", accuracy)


# --------------------------------------------------
# 3. Generative Model - Autoencoder
# --------------------------------------------------

autoencoder = Sequential([
    Flatten(input_shape=(28, 28)),

    # Encoder
    Dense(128, activation="relu"),
    Dense(32, activation="relu"),

    # Decoder
    Dense(128, activation="relu"),
    Dense(784, activation="sigmoid"),

    Reshape((28, 28))
])

autoencoder.compile(
    optimizer="adam",
    loss="mse"
)

print("\nTraining Generative Model...")

autoencoder.fit(
    x_train,
    x_train,
    epochs=3,
    batch_size=128,
    validation_split=0.1
)


# --------------------------------------------------
# 4. Image Reconstruction
# --------------------------------------------------

reconstructed = autoencoder.predict(x_test[:5])


# --------------------------------------------------
# 5. Display Original and Reconstructed Images
# --------------------------------------------------

plt.figure(figsize=(10, 4))

for i in range(5):

    # Original Image
    plt.subplot(2, 5, i + 1)
    plt.imshow(x_test[i], cmap="gray")
    plt.axis("off")
    plt.title("Original")

    # Reconstructed Image
    plt.subplot(2, 5, i + 6)
    plt.imshow(reconstructed[i], cmap="gray")
    plt.axis("off")
    plt.title("Generated")

plt.tight_layout()
plt.show()

def retrieve(query, top_k=3):
    """Return the top-k documents ranked by cosine similarity."""
    query_vector = vectorizer.transform([query])
    scores = cosine_similarity(query_vector, document_vectors)[0]

    ranked_indices = np.argsort(scores)[::-1][:top_k]

    results = []
    for index in ranked_indices:
        results.append({
            "title": documents[index]["title"],
            "text": documents[index]["text"],
            "score": float(scores[index])
        })

    return results

def grounded_answer(query, top_k=3, threshold=0.10):
    """
    Lightweight generation stage.

    The response is created only from retrieved evidence. If the
    retrieval score is too low, the system refuses to invent an
    answer. This demonstrates the grounding principle of RAG.
    """
    results = retrieve(query, top_k=top_k)

    useful = [item for item in results if item["score"] >= threshold]

    if not useful:
        return (
            "I cannot answer this question from the available "
            "knowledge base because sufficient evidence was not found."
        ), results

    evidence = " ".join(item["text"] for item in useful)

    answer = (
        "Grounded answer:\n"
        + evidence
        + "\n\nSources: "
        + ", ".join(item["title"] for item in useful)
    )

    return answer, results

Generative Adversarial Networks (GAN)

# GenAI Practical 5: Generative Adversarial Network (GAN) on MNIST
import torch
import torch.nn as nn
import torchvision
import torchvision.transforms as transforms
from torch.utils.data import DataLoader
import matplotlib.pyplot as plt

# -------------------------------
# 1. Device
# -------------------------------
device = torch.device("cuda" if torch.cuda.is_available() else "mps" if torch.backends.mps.is_available() else "cpu")

# -------------------------------
# 2. Load MNIST Dataset
# -------------------------------
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.5,), (0.5,))
])

dataset = torchvision.datasets.MNIST(
    root="./data",
    train=True,
    transform=transform,
    download=True
)

dataloader = DataLoader(dataset, batch_size=64, shuffle=True)

# -------------------------------
# 3. Generator
# -------------------------------
class Generator(nn.Module):
    def __init__(self):
        super(Generator, self).__init__()

        self.model = nn.Sequential(
            nn.Linear(100, 256),
            nn.ReLU(),

            nn.Linear(256, 512),
            nn.ReLU(),

            nn.Linear(512, 784),
            nn.Tanh()
        )

    def forward(self, x):
        return self.model(x)


# -------------------------------
# 4. Discriminator
# -------------------------------
class Discriminator(nn.Module):
    def __init__(self):
        super(Discriminator, self).__init__()

        self.model = nn.Sequential(
            nn.Linear(784, 512),
            nn.LeakyReLU(0.2),

            nn.Linear(512, 256),
            nn.LeakyReLU(0.2),

            nn.Linear(256, 1),
            nn.Sigmoid()
        )

    def forward(self, x):
        return self.model(x)


# Create models
G = Generator().to(device)
D = Discriminator().to(device)

# -------------------------------
# 5. Loss and Optimizers
# -------------------------------
criterion = nn.BCELoss()

optimizer_G = torch.optim.Adam(G.parameters(), lr=0.0002)
optimizer_D = torch.optim.Adam(D.parameters(), lr=0.0002)

# -------------------------------
# 6. Training
# -------------------------------
epochs = 10

for epoch in range(epochs):

    for real_images, _ in dataloader:

        # Flatten image: 28x28 -> 784
        real_images = real_images.view(-1, 784).to(device)

        batch_size = real_images.size(0)

        # Labels
        real_labels = torch.ones(batch_size, 1).to(device)
        fake_labels = torch.zeros(batch_size, 1).to(device)

        # --------------------------------
        # Train Discriminator
        # --------------------------------

        # Real images
        real_output = D(real_images)
        loss_real = criterion(real_output, real_labels)

        # Fake images
        noise = torch.randn(batch_size, 100).to(device)
        fake_images = G(noise)

        fake_output = D(fake_images.detach())
        loss_fake = criterion(fake_output, fake_labels)

        # Total discriminator loss
        loss_D = loss_real + loss_fake

        optimizer_D.zero_grad()
        loss_D.backward()
        optimizer_D.step()

        # --------------------------------
        # Train Generator
        # --------------------------------

        noise = torch.randn(batch_size, 100).to(device)
        fake_images = G(noise)

        output = D(fake_images)

        # Generator wants discriminator to think
        # fake images are real
        loss_G = criterion(output, real_labels)

        optimizer_G.zero_grad()
        loss_G.backward()
        optimizer_G.step()

    print(
        f"Epoch [{epoch+1}/{epochs}] "
        f"Loss D: {loss_D.item():.4f} "
        f"Loss G: {loss_G.item():.4f}"
    )

# -------------------------------
# 7. Generate Images
# -------------------------------
noise = torch.randn(16, 100).to(device)
generated_images = G(noise)

generated_images = generated_images.view(-1, 28, 28)
generated_images = generated_images.detach().cpu()

# -------------------------------
# 8. Display Generated Images
# -------------------------------
plt.figure(figsize=(6, 6))

for i in range(16):
    plt.subplot(4, 4, i + 1)
    plt.imshow(generated_images[i], cmap="gray")
    plt.axis("off")

plt.tight_layout()
plt.show()

# GenAI Practical 7: Neural style transfer
import numpy as np
import matplotlib.pyplot as plt
from PIL import Image
import torch
import torch.nn.functional as F


# ============================================================
# 1. REPRODUCIBILITY
# ============================================================

np.random.seed(42)
torch.manual_seed(42)


# ============================================================
# 2. CREATE DEMONSTRATION IMAGES
# ============================================================

def create_content_image(size=128):
    """Create a simple content image with geometric structure."""

    image = np.zeros((size, size, 3), dtype=np.float32)

    # Background
    image[:] = [0.85, 0.85, 0.85]

    # Large square
    image[25:100, 25:100] = [0.2, 0.45, 0.8]

    # Center circle
    y, x = np.ogrid[:size, :size]
    mask = (x - 64) ** 2 + (y - 64) ** 2 < 18 ** 2
    image[mask] = [0.95, 0.85, 0.2]

    return image


def create_style_image(size=128):
    """Create a colorful patterned style image."""

    image = np.zeros((size, size, 3), dtype=np.float32)

    for y in range(size):
        for x in range(size):
            image[y, x, 0] = (np.sin(x / 7.0) + 1) / 2
            image[y, x, 1] = (np.cos(y / 9.0) + 1) / 2
            image[y, x, 2] = (np.sin((x + y) / 12.0) + 1) / 2

    return image


content_np = create_content_image()
style_np = create_style_image()


# ============================================================
# 3. CONVERT TO TENSORS
# ============================================================

content = torch.tensor(
    content_np,
    dtype=torch.float32
).permute(2, 0, 1).unsqueeze(0)

style = torch.tensor(
    style_np,
    dtype=torch.float32
).permute(2, 0, 1).unsqueeze(0)


# ============================================================
# 4. STYLE REPRESENTATION
# ============================================================

def gram_matrix(tensor):
    """
    Compute Gram Matrix.

    The Gram Matrix represents correlations between feature
    channels and is commonly used for style representation.
    """

    batch, channels, height, width = tensor.shape

    features = tensor.view(
        batch,
        channels,
        height * width
    )

    gram = torch.bmm(
        features,
        features.transpose(1, 2)
    )

    return gram / (channels * height * width)


# ============================================================
# 5. CONTENT AND STYLE FEATURES
# ============================================================

content_features = content

style_features = gram_matrix(style)


# ============================================================
# 6. INITIALIZE GENERATED IMAGE
# ============================================================

generated = torch.rand_like(content)

generated.requires_grad_(True)


# ============================================================
# 7. OPTIMIZER
# ============================================================

optimizer = torch.optim.Adam(
    [generated],
    lr=0.05
)


# ============================================================
# 8. STYLE TRANSFER OPTIMIZATION
# ============================================================

epochs = 300

content_weight = 1.0

style_weight = 5.0

loss_history = []

print("=" * 60)
print("PRACTICAL 7: NEURAL STYLE TRANSFER")
print("=" * 60)

print("Image size:", "128 x 128")

print("Optimization epochs:", epochs)

print()


for epoch in range(epochs):

    optimizer.zero_grad()

    # Content loss
    content_loss = F.mse_loss(
        generated,
        content_features
    )

    # Style loss
    generated_style = gram_matrix(generated)

    style_loss = F.mse_loss(
        generated_style,
        style_features
    )

    # Total loss
    total_loss = (
        content_weight * content_loss
        + style_weight * style_loss
    )

    total_loss.backward()

    optimizer.step()

    # Keep pixels between 0 and 1
    with torch.no_grad():
        generated.clamp_(0.0, 1.0)

    loss_history.append(
        total_loss.item()
    )

    if (epoch + 1) % 50 == 0:

        print(
            f"Epoch [{epoch + 1:3d}/{epochs}] | "
            f"Content Loss: {content_loss.item():.6f} | "
            f"Style Loss: {style_loss.item():.6f} | "
            f"Total Loss: {total_loss.item():.6f}"
        )


# ============================================================
# 9. CONVERT GENERATED IMAGE
# ============================================================

generated_np = (
    generated.detach()
    .squeeze(0)
    .permute(1, 2, 0)
    .numpy()
)


# ============================================================
# 10. DISPLAY CONTENT, STYLE AND GENERATED IMAGE
# ============================================================

plt.figure(figsize=(15, 5))


plt.subplot(1, 3, 1)

plt.imshow(content_np)

plt.title("Content Image")

plt.axis("off")


plt.subplot(1, 3, 2)

plt.imshow(style_np)

plt.title("Style Image")

plt.axis("off")


plt.subplot(1, 3, 3)

plt.imshow(generated_np)

plt.title("Generated Image")

plt.axis("off")


plt.tight_layout()

plt.show()


# ============================================================
# 11. LOSS GRAPH
# ============================================================

plt.figure(figsize=(9, 5))

plt.plot(
    loss_history,
    label="Style Transfer Loss"
)

plt.title("Neural Style Transfer Optimization")

plt.xlabel("Epoch")

plt.ylabel("Loss")

plt.legend()

plt.grid(alpha=0.2)

plt.tight_layout()

plt.show()


# ============================================================
# 12. SAVE GENERATED IMAGE
# ============================================================

output_image = (
    np.clip(generated_np, 0, 1) * 255
).astype(np.uint8)

Image.fromarray(
    output_image
).save(
    "practical_7_generated_style.png"
)

print("Generated image saved as:")
print("practical_7_generated_style.png")


# ============================================================
# 13. OBSERVATION
# ============================================================

    "1. Neural Style Transfer separates content and style information."
)

print(
    "2. Content loss helps preserve the structure of the content image."
)

    "3. Style loss uses feature correlations to represent artistic style."
)

print(
    "4. The generated image is optimized to minimize both losses."
)

Generative AI vs Discriminative Models

Practical 1: Generative AI vs Discriminative Models

Problem Statement:
To study the concept of Generative Artificial Intelligence and compare Generative Models with Discriminative Models.

Aim:
To understand what Generative AI is, and to compare generative models and discriminative models using the MNIST dataset.

Theory:
Artificial Intelligence (AI) makes computers do tasks that need human intelligence. Machine Learning (ML) is a part of AI in which computers learn patterns from data.

Generative AI is a type of AI that can create new content, like text, images, music, audio, video and code. It learns the patterns of the training data and then makes new data that looks similar to it. Examples are chatbots that write text and tools that make images from a text description.

There are two main types of models:
- Discriminative models learn the boundary between classes. They answer the question “Which class does this input belong to?”. They learn P(y | x), the probability of the label y for a given input x. Examples: Logistic Regression, SVM, Decision Tree, a neural network classifier.
- Generative models learn how the data itself is made (the data distribution). They learn P(x) or P(x, y). Because of this, they can create new data. Examples: GAN, VAE (Variational Autoencoder), diffusion models and large language models (LLMs).

Comparison of the two:
- Goal: generative models create new data; discriminative models classify or predict.
- What they learn: generative models learn P(x) or P(x, y); discriminative models learn P(y | x).
- Output: a generative model gives a new sample (an image, a sentence); a discriminative model gives a label (like “digit 7”).
- Cost: generative models usually need more data and more computing power.
- Examples: GAN, VAE, LLM vs Logistic Regression, SVM, Decision Tree.

Dataset: MNIST has 70,000 images of handwritten digits (0 to 9). Each image is 28 × 28 pixels in grayscale. 60,000 images are for training and 10,000 for testing.

Discriminative part of this practical: a small neural network (Flatten → Dense 128 with ReLU → Dense 10 with softmax) learns to tell which digit is in an image.

Generative part of this practical: an autoencoder. The encoder squeezes the 784 pixels into only 32 numbers (a compressed code). The decoder rebuilds the image from these 32 numbers. The rebuilt (generated) images look like the originals. An autoencoder is a simple generative-style model; its fully generative version, the VAE, can make brand-new images.

Activation functions used: ReLU gives max(0, x). Softmax turns scores into probabilities that add up to 1. Sigmoid gives a value between 0 and 1.

Formulas:
1. Discriminative model learns: P(y | x)
2. Generative model learns: P(x) or P(x, y) = P(x | y) × P(y)
3. Bayes' rule (link between the two): P(y | x) = P(x | y) × P(y) ÷ P(x)
4. Pixel normalisation: x_new = x ÷ 255 (pixel values change from 0–255 to 0–1)
5. ReLU: f(x) = max(0, x)
6. Softmax: P(class k) = e^(zk) ÷ Σ e^(zj)
7. Cross-entropy loss (classifier): Loss = −Σ y × log(ŷ)
8. Autoencoder loss (MSE): Loss = (1 ÷ n) × Σ(x − x̂)², where x̂ is the rebuilt image
9. Accuracy = correct predictions ÷ total predictions

Applications:
- Generative models: chatbots, image and music creation, synthetic data, code writing
- Discriminative models: spam detection, face recognition, disease diagnosis, sentiment analysis

Procedure:
1. Load the MNIST dataset and normalise the pixel values to the range 0 to 1.
2. Build the discriminative model: Flatten, Dense 128 (ReLU) and Dense 10 (softmax).
3. Compile it with the Adam optimizer and sparse categorical cross-entropy loss, and train it for 3 epochs.
4. Test the classifier on the test images and print its accuracy.
5. Build the autoencoder: encoder 784 → 128 → 32 and decoder 32 → 128 → 784.
6. Train it with MSE loss, using the same images as input and output.
7. Rebuild 5 test images and show the original and generated images.

Conclusion:
We learned the difference between generative and discriminative models. The classifier (discriminative model) predicted the digits with high accuracy, and the autoencoder (generative model) rebuilt digit images that look like the originals.

Probabilistic modeling and synthetic data

Practical 2: Probabilistic modeling and synthetic data

Problem Statement:
To implement probabilistic modeling techniques and generate synthetic data samples.

Aim:
To model data using a probability distribution (the Gaussian or normal distribution) and generate synthetic data samples from it.

Theory:
Probability tells us how likely something is to happen. Its value is between 0 (impossible) and 1 (certain).

A random variable is a value that comes from a random process, like the height of a student. A probability distribution tells us which values are common and which are rare.

Probabilistic modeling means describing data with a probability distribution. Once we know the distribution, we can create new data by picking random values from it. This is called sampling.

Synthetic data is artificial data made by a computer that behaves like real data. It is useful when real data is small, costly or private (like medical data). Generative AI models are also probabilistic models: they learn the distribution of the data and then sample new data from it.

Gaussian (normal) distribution: the most common distribution. Its graph is a bell-shaped curve. It has two parameters:
- Mean (μ): the centre of the curve. In this practical μ = 50.
- Standard deviation (σ): how spread out the data is. In this practical σ = 10.

68–95–99.7 rule: about 68% of the values lie within 1σ of the mean (40 to 60 here), about 95% within 2σ (30 to 70), and about 99.7% within 3σ (20 to 80).

PDF (Probability Density Function): a curve that shows how likely each value is. The higher the curve at a point, the more likely that value is. The total area under the curve is 1.

In this practical we generate 1000 samples with mean 50 and standard deviation 10. We draw a histogram of the samples and draw the Gaussian PDF curve on top. If the histogram follows the curve, the synthetic data matches the model.

Other common distributions: uniform (all values equally likely), binomial (number of successes in fixed tries) and Poisson (number of events in a time period).

Formulas:
1. Gaussian PDF: f(x) = [1 ÷ (σ√(2π))] × e^(−(x − μ)² ÷ (2σ²))
2. Mean: μ = Σx ÷ n
3. Variance: σ² = Σ(x − μ)² ÷ n
4. Standard deviation: σ = √variance
5. Z-score: z = (x − μ) ÷ σ
6. 68–95–99.7 rule: P(μ − σ < X < μ + σ) ≈ 0.68, P(μ − 2σ < X < μ + 2σ) ≈ 0.95, P(μ − 3σ < X < μ + 3σ) ≈ 0.997
7. Area under any PDF = 1

Applications:
- Making synthetic data for testing and training models
- Data privacy (sharing realistic but fake data)
- Simulations in science, engineering and finance
- The base idea behind generative models like VAEs and diffusion models

Procedure:
1. Set the parameters of the Gaussian distribution: mean = 50 and standard deviation = 10.
2. Generate 1000 synthetic samples using np.random.normal.
3. Make 1000 points between 10 and 90 and find the Gaussian PDF value of each point using norm.pdf.
4. Draw a histogram of the synthetic data (30 bins, density = True).
5. Draw the PDF curve on the same graph and compare it with the histogram.

Conclusion:
We modelled data with a Gaussian distribution and generated 1000 synthetic samples from it. The histogram of the samples closely follows the Gaussian curve, which shows that the synthetic data matches the probability model.

Transformer language models and text generation

Practical 3: Transformer language models and text generation

Problem Statement:
To study the architecture of transformer-based language models and generate text using pretrained models.

Aim:
To study the architecture of transformer-based language models and generate text using the pretrained GPT-2 model.

Theory:
A language model predicts the next word (token) in a piece of text. For example, after “The sky is”, a likely next word is “blue”.

The Transformer is a deep learning architecture introduced in 2017 in the paper “Attention Is All You Need”. Older models (RNNs) read words one by one. A Transformer uses attention to look at all the words at once, so it trains faster and understands long text better.

Main parts of a Transformer:
- Tokenization: the text is broken into small pieces called tokens (words or parts of words).
- Embeddings: each token is changed into a list of numbers (a vector).
- Positional encoding: adds the position of each token, because attention alone does not know the word order.
- Self-attention: each word looks at all the other words and decides which ones matter for its meaning. For example, in “The cat drank milk because it was hungry”, attention links “it” to “cat”.
- Multi-head attention: many attention “heads” run side by side, and each one looks at a different kind of relation.
- Feed-forward network: a small neural network applied to each token after attention.
- Residual connections and layer normalisation: help information flow through many layers and keep training stable.

Types of transformer models:
- Encoder-only (like BERT): good for understanding text, such as classification.
- Decoder-only (like GPT): good for generating text.
- Encoder-decoder (like T5): good for translation and summarisation.

GPT-2 (Generative Pre-trained Transformer 2) is a decoder-only model made by OpenAI in 2019. It was pretrained on a large amount of internet text. It is autoregressive: it predicts one token at a time, adds it to the text, and repeats until enough tokens are made.

Text generation settings: max_new_tokens (how many new tokens to make), temperature (low = safe and repetitive, high = creative and random), and top-k / top-p (choose only from the most likely tokens).

The Hugging Face pipeline loads the tokenizer and the model together, so we can generate text in only a few lines of code.

Formulas:
1. Self-attention: Attention(Q, K, V) = softmax(Q × K^T ÷ √dk) × V (Q = query, K = key, V = value, dk = size of the key vector)
2. Softmax: softmax(zi) = e^(zi) ÷ Σ e^(zj)
3. Multi-head attention = join(head1, head2, …, headh) × W
4. Positional encoding: PE(pos, 2i) = sin(pos ÷ 10000^(2i/d)) and PE(pos, 2i+1) = cos(pos ÷ 10000^(2i/d))
5. Autoregressive model: P(w1, w2, …, wn) = P(w1) × P(w2 | w1) × … × P(wn | w1 … wn−1)
6. Temperature: P(token i) = e^(zi/T) ÷ Σ e^(zj/T) (T = temperature)
7. Perplexity = e^(average loss) (lower is better)

Applications:
- Chatbots and virtual assistants
- Writing help, summaries and translation
- Code generation
- Question answering

Procedure:
1. Install the transformers and torch libraries.
2. Import pipeline from the transformers library.
3. Load the pretrained GPT-2 model using pipeline("text-generation", model="gpt2").
4. Give the prompt “Artificial Intelligence is” and generate 50 new tokens.
5. Print the prompt and the generated text.
6. Try more prompts and compare the generated outputs.

Conclusion:
We studied the main parts of the Transformer architecture and used the pretrained GPT-2 model to generate text. GPT-2 continued each prompt by predicting one token at a time, and different prompts gave different text.

Prompt engineering and RAG to reduce hallucination

Practical 4: Prompt engineering and RAG to reduce hallucination

Problem Statement:
To design effective prompts and implement Retrieval Augmented Generation to reduce hallucination in large language models.

Aim:
To write effective prompts and build a simple Retrieval Augmented Generation (RAG) system that reduces hallucination.

Theory:
A Large Language Model (LLM) is a very big transformer model trained on a huge amount of text. A prompt is the input or instruction that we give to the model.

Prompt engineering means writing prompts in a clear and smart way to get better and more consistent answers.

Principles of a good prompt:
- Give a role, like “You are a computer science tutor”.
- Give a clear task.
- Say the output format, like “use exactly 4 bullet points”.
- Add limits, like length or style.
- Give examples when needed.

Types of prompts:
- Zero-shot: only the instruction, with no examples.
- Few-shot: a few solved examples are given first, then the new question.
- Role prompt: the model is asked to act as someone, like a teacher.
- Chain-of-thought: the model is asked to show its steps before the final answer.

Hallucination: sometimes an LLM gives an answer that sounds right but is false or made up. This happens because the model predicts likely words, not checked facts, and its knowledge can be old or missing.

RAG (Retrieval Augmented Generation) reduces hallucination. Before answering, the system searches a knowledge base (a set of documents) for relevant information and gives it to the model as context. The answer is then based on real evidence. This is called grounding.

RAG pipeline in this practical:
1. The user asks a question (query).
2. The query and the documents are changed into vectors using TF-IDF.
3. Cosine similarity finds how close each document is to the query.
4. The top-k documents are selected (here k = 3).
5. The answer is built only from these documents, and the sources are shown.
6. If no document is similar enough (score below a threshold), the system refuses to answer instead of making something up.

In real systems, embeddings from neural networks and a vector database are used instead of TF-IDF, and the retrieved text is given to an LLM to write the final answer.

Formulas:
1. TF = (times the word appears in a document) ÷ (total words in the document)
2. IDF = log(N ÷ number of documents that contain the word), where N = total documents (basic form)
3. TF-IDF = TF × IDF
4. Cosine similarity: cos(θ) = (A · B) ÷ (||A|| × ||B||) (1 = very similar, 0 = not related)
5. Top-k retrieval: choose the k documents with the highest cosine similarity
6. Grounding rule: answer only if similarity ≥ threshold (0.10 in the main test, 0.25 in the hallucination test), otherwise refuse

Applications:
- Company chatbots that answer from their own documents
- Customer support systems
- Search engines with AI answers
- Study, legal and medical assistants that must give correct facts

Procedure:
1. Write a weak prompt, a clear structured prompt and a few-shot prompt, and compare them.
2. Create a small knowledge base of 7 documents about AI topics.
3. Change the documents into TF-IDF vectors (with English stop words removed).
4. Write a retrieve function that finds the top 3 documents for a query using cosine similarity.
5. Ask “How does RAG help reduce hallucinations?”, show the retrieved documents with their scores, and draw a bar graph of the scores.
6. Build a grounded answer using only the documents whose score is above the threshold, and show the sources.
7. Ask an unrelated question (“Who invented quantum computing in the year 1890?”) and check that the system refuses to answer.

Conclusion:
Clear, structured and few-shot prompts give better instructions to a model. The RAG system found the right documents using TF-IDF and cosine similarity and answered only from them. For an unrelated question it refused to answer, which shows how RAG reduces hallucination.

Generative Adversarial Networks (GAN)

Practical 5: Generative Adversarial Networks (GAN)

Problem Statement:
To study the architecture and training process of Generative Adversarial Networks.

Aim:
To study the architecture and training process of a Generative Adversarial Network (GAN) and use it to generate handwritten digit images.

Theory:
A GAN (Generative Adversarial Network) was introduced by Ian Goodfellow in 2014. It has two neural networks that compete with each other, like players in a game.

- Generator (G): takes random noise (here a list of 100 random numbers) and turns it into a fake image. Its goal is to make images so real that the discriminator is fooled.
- Discriminator (D): takes an image and says whether it is real (from the dataset) or fake (from the generator). It gives a probability between 0 and 1 (1 = real, 0 = fake).

Simple example: the generator is like a person making fake currency notes, and the discriminator is like the police checking them. Both keep getting better, until the fake notes look real.

Architecture in this practical:
- Generator: 100 → 256 (ReLU) → 512 (ReLU) → 784 (Tanh). The 784 outputs form a 28 × 28 image. Tanh gives values between −1 and 1.
- Discriminator: 784 → 512 (LeakyReLU) → 256 (LeakyReLU) → 1 (Sigmoid). Sigmoid gives the probability that the image is real.

Training process (for each batch of 64 images):
1. Train D on real images with label 1 and on fake images with label 0.
2. Add both losses and update D.
3. Make new fake images and pass them to D, but this time use label 1 (real). This loss teaches G how to fool D.
4. Update G. Repeat this for all batches, for 10 epochs.

The MNIST images are normalised to the range −1 to 1, so that they match the Tanh output of the generator.

Loss used: Binary Cross-Entropy (BCE). Optimizer: Adam with learning rate 0.0002 for both networks.

Common problems: mode collapse (the generator makes only a few kinds of images), unstable training, and vanishing gradients when D becomes much stronger than G.

Types of GANs: DCGAN (uses convolution layers), Conditional GAN (makes images of a chosen class), CycleGAN (changes style, like photo to painting) and StyleGAN (very realistic faces).

Formulas:
1. GAN objective (minimax game): min over G, max over D of V(D, G) = E[log D(x)] + E[log(1 − D(G(z)))]
2. Here x = real image, z = random noise, G(z) = fake image, D(·) = probability of being real
3. Binary cross-entropy: BCE = −[y × log(p) + (1 − y) × log(1 − p)]
4. Discriminator loss: Loss_D = BCE(D(x), 1) + BCE(D(G(z)), 0)
5. Generator loss: Loss_G = BCE(D(G(z)), 1)
6. Normalisation: x_new = (x − 0.5) ÷ 0.5 (changes 0 to 1 into −1 to 1)
7. ReLU = max(0, x); LeakyReLU = x if x > 0, else 0.2x; Tanh = (e^x − e^(−x)) ÷ (e^x + e^(−x)); Sigmoid = 1 ÷ (1 + e^(−x))

Applications:
- Making realistic images and faces
- Data augmentation (making extra training data)
- Image-to-image translation and style change
- Improving image quality (super-resolution)

Procedure:
1. Select the device (GPU, Apple MPS or CPU).
2. Load the MNIST dataset, change the images to tensors, normalise them to −1 to 1, and make batches of 64.
3. Build the Generator (100 → 256 → 512 → 784) and the Discriminator (784 → 512 → 256 → 1).
4. Set the BCE loss and an Adam optimizer (learning rate 0.0002) for each network.
5. For each batch, train the discriminator on real and fake images, then train the generator to fool the discriminator.
6. Train for 10 epochs and print the losses of D and G after each epoch.
7. Give 16 random noise vectors to the generator and show the 16 generated digit images in a 4 × 4 grid.

Conclusion:
We studied the generator–discriminator structure of a GAN and trained it on MNIST. After training, the generator produced new images that look like handwritten digits, which shows how adversarial training helps a model create realistic data.

Image generation with Stable Diffusion

Practical 6: Image generation with Stable Diffusion

Problem Statement:
To generate images using diffusion-based generative models such as Stable Diffusion.

Aim:
To generate an image from a text prompt using the Stable Diffusion model.

Theory:
Diffusion models are generative models that create images by slowly removing noise. Many modern image generators are based on them.

A diffusion model has two processes:
- Forward process (adding noise): a real image is covered with a little random noise at each step, until after many steps it becomes pure noise.
- Reverse process (removing noise): a neural network learns to remove the noise step by step. To make a new image, we start from pure noise and remove the noise again and again until a clear image appears.

During training, the model is shown a noisy image and learns to predict the noise that was added. A model that can predict the noise well can clean any noisy image.

Stable Diffusion is a latent diffusion model. Instead of working on the full-size image, it works on a much smaller compressed version of it, called the latent space. This makes it faster and lighter.

Main parts of Stable Diffusion:
- Text encoder (CLIP): changes the text prompt into numbers (embeddings) that describe its meaning.
- U-Net: the main network. It predicts and removes the noise in the latent space, guided by the text embeddings through cross-attention.
- Scheduler: decides how much noise to remove at each step.
- VAE decoder: changes the final clean latent back into a full image (512 × 512 pixels for version 1.5).
- Safety checker: checks the image for unsafe content.

Important settings:
- Prompt: the text description of the image we want.
- num_inference_steps: the number of denoising steps (here 15). More steps usually give more detail but take more time.
- guidance_scale: how strongly the image should follow the prompt (here 7.5). A higher value follows the prompt more closely but can look less natural.
- Device and data type: on a GPU, float16 is used to save memory and time; on a CPU or Apple MPS, float32 is used.

enable_attention_slicing() calculates attention in small parts, so the model needs less memory.

Formulas:
1. Forward step: q(xt | xt−1) = N(√(1 − βt) × xt−1, βt × I) (βt = small amount of noise added at step t)
2. Jump to any step: xt = √(ᾱt) × x0 + √(1 − ᾱt) × ε, where ε = random noise and ᾱt = (1 − β1) × (1 − β2) × … × (1 − βt)
3. Training loss: L = || ε − εθ(xt, t, text) ||² (the model εθ predicts the added noise)
4. Classifier-free guidance: ε = ε_uncond + s × (ε_cond − ε_uncond) (s = guidance scale = 7.5)
5. Latent size: a 512 × 512 × 3 image becomes a 64 × 64 × 4 latent, which is 48 times smaller

Applications:
- Art, posters and illustrations
- Product and fashion design ideas
- Image editing (inpainting) and upscaling
- Concept art for games and films

Procedure:
1. Import torch, matplotlib and StableDiffusionPipeline from the diffusers library.
2. Select the device (CUDA GPU, Apple MPS or CPU) and the data type (float16 on GPU, float32 otherwise).
3. Load the pretrained Stable Diffusion v1.5 model and move it to the device.
4. Turn on attention slicing to save memory.
5. Write a text prompt that describes the image.
6. Generate the image with 15 inference steps and a guidance scale of 7.5.
7. Show the image using matplotlib and save it as generated_image.png.

Conclusion:
We used the Stable Diffusion model to turn a text prompt into an image. The model started from random noise and removed it step by step, guided by the prompt, to create a detailed image that matches the description.

Artistic content with Neural Style Transfer

Practical 7: Artistic content with Neural Style Transfer

Problem Statement:
To generate artistic content using neural style transfer and GAN-based models.

Aim:
To generate artistic images using Neural Style Transfer (NST), which mixes the content of one image with the style of another.

Theory:
Neural Style Transfer (NST) takes two images: a content image (what to draw, like shapes and objects) and a style image (how to draw it, like colours, textures and brush strokes). It creates a new image that has the content of the first image and the style of the second. It was introduced by Gatys and others in 2015.

Content means the structure of the image, like where the objects and edges are. Style means the colours, textures and patterns that repeat across the image.

Content loss measures how different the generated image is from the content image. Style loss measures how different the styles of the generated image and the style image are.

Gram matrix: style is measured with a Gram matrix. It shows how much the different feature channels (here the red, green and blue channels) appear together. It keeps the colour and texture information but forgets where things are, which is exactly what style means.

Total loss = content weight × content loss + style weight × style loss. Here the content weight is 1 and the style weight is 5.

How the image is made: we start from a random noise image and treat its pixels as the values to learn. In each step the losses are calculated, the gradients are found with backpropagation, and the Adam optimizer changes the pixels a little (learning rate 0.05). Pixel values are kept between 0 and 1. After 300 steps, the image has the structure of the content image with the colours and patterns of the style image.

Note on this practical: to keep it simple and fast, it uses two small 128 × 128 images made by code (a square-and-circle content image and a colourful wave-pattern style image) and compares the raw colour channels. The original NST method uses features from a pretrained CNN called VGG-19, which gives richer artistic results.

GAN-based art: GANs can also create art. CycleGAN can turn photos into paintings (for example, into a Monet-style painting) without needing matching image pairs. StyleGAN can make new, realistic faces and artworks. Unlike NST, a GAN learns a style from many images, and after training it can apply it very quickly.

Formulas:
1. Content loss: L_content = (1 ÷ N) × Σ(G − C)² (G = generated image, C = content image)
2. Gram matrix: Gram(i, j) = Σk F(i, k) × F(j, k) ÷ (channels × height × width) (F = values of channels i and j)
3. Style loss: L_style = (1 ÷ N) × Σ(Gram(G) − Gram(S))² (S = style image)
4. Total loss: L_total = α × L_content + β × L_style (here α = 1 and β = 5)
5. Update rule (gradient descent idea): pixels = pixels − learning rate × ∂L_total/∂pixels
6. Clamp: pixel = min(max(pixel, 0), 1)

Applications:
- Turning photos into paintings
- Photo filters in art and design apps
- Creating textures for games and films
- Making creative posters and wallpapers

Procedure:
1. Set the random seeds so that the results are the same every time.
2. Create a 128 × 128 content image (grey background, blue square, yellow circle) and a colourful wave-pattern style image.
3. Convert both images into PyTorch tensors.
4. Write a function for the Gram matrix and find the Gram matrix of the style image.
5. Start the generated image from random noise and allow its pixels to be learned (requires_grad).
6. Use the Adam optimizer (learning rate 0.05) for 300 steps: find the content loss, style loss and total loss, backpropagate, update the pixels and keep them between 0 and 1.
7. Print the losses every 50 steps.
8. Show the content, style and generated images side by side, and draw the loss graph.
9. Save the generated image as practical_7_generated_style.png.

Conclusion:
Neural Style Transfer was done using PyTorch. The content loss kept the structure of the content image, and the Gram-matrix-based style loss added the colours and patterns of the style image. The final image combines both.

Train, fine-tune and deploy an open-source model

Practical 8: Train, fine-tune and deploy an open-source model

Problem Statement:
To train, fine-tune, and deploy open-source generative AI models.

Aim:
To fine-tune an open-source generative AI model (DistilGPT-2) on our own text data, save it, and deploy it locally for text generation.

Theory:
Open-source models are AI models whose weights are free to download and use, like GPT-2, DistilGPT-2, LLaMA, Mistral and Stable Diffusion. Hugging Face is a popular platform that hosts these models and gives easy libraries to use them.

Three stages in the life of a model:
- Training (pre-training): the model learns from a very large amount of general data. This needs a lot of time and computing power.
- Fine-tuning: a pretrained model is trained a little more on a small, task-specific dataset, so that it adapts to that task or style. This is a type of transfer learning and is much cheaper than training from scratch.
- Deployment: the trained model is saved and made ready for use, for example through a local pipeline, an API or a web app.

DistilGPT-2 is a smaller and faster version of GPT-2, made with knowledge distillation (a small “student” model learns from a big “teacher” model). It has about 82 million parameters, while the smallest GPT-2 has about 124 million.

Causal language modelling: the model learns to predict the next token from the tokens before it. That is why mlm = False is used (MLM, masked language modelling, is the method used by BERT).

Fine-tuning in this practical:
- Dataset: 8 sentences about AI, repeated 10 times (80 examples).
- Tokenization: each sentence is changed into tokens and cut or padded to a length of 64. GPT-2 has no padding token, so the end-of-sentence token is used for padding.
- Training settings: 2 epochs, batch size 2, a log every 10 steps.
- The Hugging Face Trainer runs the training loop: forward pass, loss, backpropagation and weight update.

Important training words:
- Epoch: one full pass over the training data.
- Batch size: the number of examples used in one update.
- Learning rate: how big each weight update is.
- Loss: how wrong the model is; it should go down during training.
- Overfitting: with a very small dataset the model may just memorise the sentences, and we can see this in the generated text.

Deployment in this practical: the fine-tuned model and tokenizer are saved in a folder and loaded with a text-generation pipeline. The prompt “Generative artificial intelligence” is given, and the model generates up to 50 new tokens using sampling with temperature 0.8.

Other ways to fine-tune big models cheaply: LoRA / PEFT (train only a few extra weights) and quantisation (store the weights with fewer bits).

Formulas:
1. Causal LM loss: L = −(1 ÷ N) × Σ log P(token t | tokens before t)
2. Perplexity = e^L (lower means the model is less confused)
3. Weight update (gradient descent): w = w − learning rate × ∂L/∂w
4. Training steps = (number of examples ÷ batch size) × epochs = (80 ÷ 2) × 2 = 80 steps
5. Temperature sampling: P(token i) = e^(zi/T) ÷ Σ e^(zj/T), with T = 0.8
6. Knowledge distillation (idea): student loss = loss on the true data + loss for matching the teacher's outputs

Applications:
- Custom chatbots for a company or a college
- Domain-specific writing (medical, legal, education)
- Running AI models privately on your own computer
- Quick prototypes of AI products

Procedure:
1. Select the device (GPU or CPU).
2. Load the DistilGPT-2 tokenizer and model from Hugging Face, and set the padding token to the end-of-sentence token.
3. Make a small dataset of 8 AI sentences repeated 10 times and convert it into a Hugging Face Dataset.
4. Tokenize the text with truncation and padding to 64 tokens.
5. Set the training arguments (2 epochs, batch size 2, no checkpoint saving) and the data collator for causal language modelling (mlm = False).
6. Fine-tune the model using the Trainer.
7. Save the fine-tuned model and tokenizer in a folder.
8. Deploy the model locally with a text-generation pipeline and generate text for the prompt “Generative artificial intelligence”.

Conclusion:
We fine-tuned the open-source DistilGPT-2 model on our own small dataset, saved it and deployed it locally with a pipeline. The generated text uses the sentences from our training data, which shows that fine-tuning changed the model's behaviour. Because the dataset is very small, the model also memorises it, so a bigger dataset is needed for real use.
