df.info()

df = df.drop_duplicates()                                # remove duplicates

    if df[col].dtype.kind in 'if':

print("\nShape after cleaning:", df.shape)

# Exp 2: Pearson correlation on PIMA Indians Diabetes dataset
import numpy as np
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
from scipy.stats import pearsonr

# Dataset path - download ke baad yahan apna path likho
# (Mac: /Users/<name>/Downloads/file.csv | Windows: C:\Users\<name>\Downloads\file.csv)
DATA_PATH = r"/path/to/diabetes.csv"

df = pd.read_csv(DATA_PATH)
print(df.head())

# Pearson formula: r = sum((x - x_mean)(y - y_mean)) / sqrt(sum((x - x_mean)^2) * sum((y - y_mean)^2))
def pearson(x, y):
    xd, yd = x - x.mean(), y - y.mean()
    return (xd * yd).sum() / np.sqrt((xd ** 2).sum() * (yd ** 2).sum())

print("\nGlucose vs Outcome (manual formula):", round(pearson(df['Glucose'], df['Outcome']), 4))
r, p = pearsonr(df['Glucose'], df['Outcome'])
print(f"Glucose vs Outcome (scipy): r = {r:.4f}, p-value = {p:.4e}")

# Full correlation matrix
corr = df.corr(method='pearson')
print("\nCorrelation of each feature with Outcome:\n", corr['Outcome'].sort_values(ascending=False))

plt.figure(figsize=(9, 7))
sns.heatmap(corr, annot=True, fmt='.2f', cmap='coolwarm')
plt.title('Pearson Correlation Matrix - PIMA Diabetes')
plt.tight_layout()
plt.show()

sns.regplot(x='Glucose', y='BMI', data=df, line_kws={'color': 'red'})
plt.title(f"Glucose vs BMI (r = {df['Glucose'].corr(df['BMI']):.2f})")
plt.show()

Principal Component Analysis (PCA) to reduce features

# Exp 3: Principal Component Analysis (PCA) on Mall Customer data
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA

# Dataset path - download ke baad yahan apna path likho
# (Mac: /Users/<name>/Downloads/file.csv | Windows: C:\Users\<name>\Downloads\file.csv)
DATA_PATH = r"/path/to/Mall_Customers.csv"

df = pd.read_csv(DATA_PATH)
print(df.head())

# Drop ID, encode Gender, scale features
X = pd.get_dummies(df.drop(columns='CustomerID'), drop_first=True).astype(float)
print("Features used:", list(X.columns))
X_scaled = StandardScaler().fit_transform(X)

# PCA with all components -> check explained variance
pca_full = PCA().fit(X_scaled)
print("\nExplained variance ratio:", pca_full.explained_variance_ratio_.round(3))
print("Cumulative variance     :", pca_full.explained_variance_ratio_.cumsum().round(3))

plt.plot(range(1, X.shape[1] + 1), pca_full.explained_variance_ratio_.cumsum(), marker='o')
plt.xlabel('Number of components'); plt.ylabel('Cumulative explained variance')
plt.title('Scree Plot'); plt.grid(); plt.show()

# Reduce to 2 components
pca = PCA(n_components=2)
X_pca = pca.fit_transform(X_scaled)
print("\nOriginal shape:", X.shape, "-> Reduced shape:", X_pca.shape)
print("Variance kept by 2 PCs:", round(pca.explained_variance_ratio_.sum() * 100, 2), "%")
print("\nPC loadings:\n", pd.DataFrame(pca.components_, columns=X.columns, index=['PC1', 'PC2']).round(3))

plt.scatter(X_pca[:, 0], X_pca[:, 1], c=df['Spending Score (1-100)'], cmap='viridis')
plt.colorbar(label='Spending Score')
plt.xlabel('PC1'); plt.ylabel('PC2'); plt.title('Customers in 2D (PCA)')
plt.show()

# Exp 7: End-to-end predictive analytics on finance data (build -> evaluate -> visualize)
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.pipeline import make_pipeline
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score, classification_report, ConfusionMatrixDisplay

# Dataset path - download ke baad yahan apna path likho
# (Mac: /Users/<name>/Downloads/file.csv | Windows: C:\Users\<name>\Downloads\file.csv)
DATA_PATH = r"/path/to/Finance_data.csv"

df = pd.read_csv(DATA_PATH)
TARGET = 'Avenue'                       # target column (loan data ho to e.g. 'Loan_Status' / 'default')
if TARGET not in df.columns:
    TARGET = df.columns[-1]

# 1. Data understanding & cleaning
print(df.shape); print(df.head())
print("Missing:", df.isnull().sum().sum(), "| Duplicates:", df.duplicated().sum())
df = df.drop_duplicates().dropna(axis=1, how='all')
for col in df.columns:
    df[col] = df[col].fillna(df[col].mode()[0])

# 2. EDA
sns.countplot(y=df[TARGET].astype(str)); plt.title(f'Target: {TARGET}'); plt.tight_layout(); plt.show()

# 3. Encoding
for col in df.select_dtypes(exclude='number').columns:
    df[col] = LabelEncoder().fit_transform(df[col].astype(str))
X, y = df.drop(columns=TARGET), df[TARGET]
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.25, random_state=42)

# 4. Model building & evaluation
models = {'Logistic Regression': make_pipeline(StandardScaler(), LogisticRegression(max_iter=2000)),
          'Random Forest': RandomForestClassifier(n_estimators=200, random_state=42)}
for name, model in models.items():
    model.fit(X_train, y_train)
    pred = model.predict(X_test)
    print(f"\n{name} Accuracy: {accuracy_score(y_test, pred):.3f}")
    print(classification_report(y_test, pred, zero_division=0))

# 5. Visualization
rf = models['Random Forest']
ConfusionMatrixDisplay.from_predictions(y_test, rf.predict(X_test), cmap='Blues')
plt.title('Random Forest - Confusion Matrix'); plt.show()

plt.title('Top 10 Important Features'); plt.tight_layout(); plt.show()

# Exp 8: Twitter sentiment analysis (Positive / Negative / Neutral / Irrelevant)
import re
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score, classification_report, ConfusionMatrixDisplay

# Dataset path - download ke baad yahan apna path likho
# (Mac: /Users/<name>/Downloads/file.csv | Windows: C:\Users\<name>\Downloads\file.csv)
TRAIN_PATH = r"/path/to/twitter_training.csv"
TEST_PATH = r"/path/to/twitter_validation.csv"

cols = ['id', 'entity', 'sentiment', 'text']
train = pd.read_csv(TRAIN_PATH, header=None, names=cols).dropna()
test = pd.read_csv(TEST_PATH, header=None, names=cols).dropna()
print(train.head())

# Text cleaning
def clean(t):
    t = re.sub(r'http\S+|@\w+|#|[^a-zA-Z\s]', ' ', str(t).lower())
    return re.sub(r'\s+', ' ', t).strip()

train['clean'] = train['text'].apply(clean)
test['clean'] = test['text'].apply(clean)

train['sentiment'].value_counts().plot.bar(color=['tab:green', 'tab:red', 'tab:gray', 'tab:blue'])
plt.title('Sentiment Distribution'); plt.ylabel('Tweets'); plt.show()

# TF-IDF + Logistic Regression
vec = TfidfVectorizer(max_features=20000, ngram_range=(1, 2), stop_words='english')
X_train = vec.fit_transform(train['clean'])
X_test = vec.transform(test['clean'])

model = LogisticRegression(max_iter=1000)
model.fit(X_train, train['sentiment'])
pred = model.predict(X_test)

print("\nAccuracy:", round(accuracy_score(test['sentiment'], pred) * 100, 2), "%")
print(classification_report(test['sentiment'], pred))
ConfusionMatrixDisplay.from_predictions(test['sentiment'], pred, cmap='Blues', xticks_rotation=45)
plt.title('Confusion Matrix'); plt.tight_layout(); plt.show()

# Try your own tweets
for tweet in ["I love this game, it's amazing!", "Worst update ever, totally broken"]:
    print(tweet, "->", model.predict(vec.transform([clean(tweet)]))[0])

Time series analysis & forecasting with ARIMA

# Exp 9: Time series analysis & forecasting with ARIMA (transportation data)
import warnings
import pandas as pd
import matplotlib.pyplot as plt
from statsmodels.tsa.stattools import adfuller
from statsmodels.tsa.arima.model import ARIMA
from statsmodels.graphics.tsaplots import plot_acf, plot_pacf
from sklearn.metrics import mean_absolute_error, mean_squared_error
warnings.filterwarnings('ignore')

# Dataset path - download ke baad yahan apna path likho
# (Mac: /Users/<name>/Downloads/file.csv | Windows: C:\Users\<name>\Downloads\file.csv)
DATA_PATH = r"/path/to/transport.csv"

df = pd.read_csv(DATA_PATH)
print(df.head())
DATE_COL = None                         # e.g. 'Date'   (None = auto detect)
VALUE_COL = None                        # e.g. 'Vehicles' (None = auto detect)

if DATE_COL is None:
    DATE_COL = next(c for c in df.select_dtypes(exclude='number').columns
                    if pd.to_datetime(df[c], errors='coerce').notna().mean() > 0.8)
if VALUE_COL is None:
    nums = [c for c in df.select_dtypes('number').columns if 'id' not in c.lower()]
    VALUE_COL = max(nums, key=lambda c: df[c].nunique())
print("Date column:", DATE_COL, "| Value column:", VALUE_COL)

# Prepare series
df[DATE_COL] = pd.to_datetime(df[DATE_COL], errors='coerce')
ts = df.dropna(subset=[DATE_COL]).groupby(DATE_COL)[VALUE_COL].sum().sort_index()
step = ts.index.to_series().diff().median()
if step < pd.Timedelta('1D'):           # hourly/minute data -> daily total
    ts, step = ts.resample('D').sum(), pd.Timedelta('1D')
ts = ts.interpolate()

ts.plot(figsize=(12, 4), title=f'{VALUE_COL} over time'); plt.show()

# Stationarity check (ADF test)
p_value = adfuller(ts)[1]
print("ADF p-value:", round(p_value, 4), "->", "Stationary" if p_value < 0.05 else "Not stationary (use d=1)")
d = 0 if p_value < 0.05 else 1
lags = min(30, len(ts) // 2 - 1)
plot_acf(ts.diff().dropna() if d else ts, lags=lags); plot_pacf(ts.diff().dropna() if d else ts, lags=lags); plt.show()

# Train / test split + ARIMA
split = int(len(ts) * 0.8)
train, test = ts[:split], ts[split:]
model = ARIMA(train, order=(2, d, 2)).fit()
print(model.summary())
pred = model.forecast(len(test))
pred.index = test.index
print("MAE :", round(mean_absolute_error(test, pred), 3))
print("RMSE:", round(mean_squared_error(test, pred) ** 0.5, 3))

# Forecast next 30 periods
future = ARIMA(ts, order=(2, d, 2)).fit().forecast(30)
future.index = pd.date_range(ts.index[-1], periods=31, freq=pd.infer_freq(ts.index) or step)[1:]
print("\nNext 30 forecasts:\n", future.round(2))

plt.figure(figsize=(12, 5))
plt.plot(train, label='Train'); plt.plot(test, label='Test')
plt.plot(pred, label='ARIMA Prediction'); plt.plot(future, '--', label='Future Forecast')
plt.legend(); plt.title('ARIMA Forecast'); plt.show()

Problem Statement:
Download a real-world dataset (Healthcare / Finance) from an open data repository and detect missing values, duplicate records, and inconsistent entries.

Aim:
To find and fix empty values, repeated rows and wrong entries in a healthcare dataset.

Theory:
Real-world data is often messy. Before we use it, we must check it and clean it.

Missing values are empty cells in the data. Duplicate records are rows that are repeated. Inconsistent entries are values written in different ways (like “Male” and “male”) or values that are not possible (like a negative bill amount).

To fix these problems, we remove repeated rows, correct or remove wrong values, and fill the empty cells. Number columns are filled with the median (the middle value). Text columns are filled with the mode (the most common value).

If the data is bad, the result will also be bad. So data cleaning is the first step of any analysis.

Procedure:
1. Load the dataset and look at its rows and columns.
2. Count the empty cells in each column.
3. Count the repeated rows.
4. Find wrong entries, like names in mixed case, negative bill amounts and wrong dates.
5. Clean the data: fix the names, remove repeated and wrong rows, and fill the empty cells.
6. Check the data again to make sure it is clean.

Conclusion:
We found the empty values, repeated rows and wrong entries, and fixed them. The data is now clean and ready to use.

Practical 2: Pearson correlation between variables

Problem Statement:
Using a suitable real-world dataset (e.g., PIMA Indians Diabetes), implement Pearson correlation in Python to study relationships between variables.

Aim:
To find how the columns of the diabetes dataset are related to each other using Pearson correlation.

Theory:
Correlation tells us how two things change together. Pearson correlation gives this as a number called r.

The value of r is between -1 and +1. If r is near +1, both values go up together. If r is near -1, one value goes up when the other goes down. If r is near 0, there is no clear relation.

Formula: r = Σ(x − x̄)(y − ȳ) / √[Σ(x − x̄)² × Σ(y − ȳ)²], where x̄ and ȳ are the average values of x and y.

If the p-value is less than 0.05, the relation is real and not by chance. Correlation only shows a relation. It does not prove that one thing causes the other.

Procedure:
1. Load the diabetes dataset.
2. Find r between Glucose and Outcome using the formula, and check it with a Python function.
3. Find r for every pair of columns (this is called the correlation matrix).
4. See which columns are most related to diabetes.
5. Draw a heatmap (a coloured table) to show all the values.

Conclusion:
We found how the columns are related to each other. Glucose has the strongest relation with diabetes.

Practical 5: SVM: is a tumour benign or malignant?

Problem Statement:
Using the Cancer Dataset, implement a Support Vector Machine (SVM) model to predict whether a tumour is benign or malignant based on given diagnostic features.

Aim:
To predict whether a tumour is benign (not cancer) or malignant (cancer) using SVM.

Theory:
SVM (Support Vector Machine) is used to put data into groups, called classes. Here the two classes are benign and malignant.

SVM draws a line that separates the two groups. It chooses the line with the widest gap between the groups. This gap is called the margin. The points closest to the line are called support vectors.

If the groups cannot be separated by a straight line, SVM uses a kernel. A kernel helps it draw a curved boundary.

We check the model with accuracy and a confusion matrix (a table of right and wrong answers). Here, missing a cancer case is the worst mistake.

Procedure:

Time series analysis & forecasting with ARIMA

Practical 9: Time series analysis & forecasting with ARIMA

Problem Statement:
Using a suitable real-world time series dataset related to transportation traffic (such as road traffic volume, metro passenger count, railway ridership, or air traffic statistics), perform time series analysis (using ARIMA) and forecasting.

Aim:
To study transport traffic data over time and predict future traffic using ARIMA.

Theory:
A time series is data collected over time, like the number of vehicles every day. The order of the data is very important.

A time series can have a trend (going up or down over a long time) and seasonality (a pattern that repeats, like less traffic every Sunday).

ARIMA is a model used to predict future values. It has three parts: p uses past values, d makes the data steady by taking differences, and q uses past errors.

ARIMA works best when the data is stationary, which means its average does not change over time. We check this with the ADF test. If the p-value is less than 0.05, the data is stationary.

We never shuffle time series data. We train the model on the older part and test it on the newer part.

Procedure:
1. Load the dataset and find the date and value columns.
2. Put the data in date order. If the data is hourly, change it to daily totals.
3. Draw a graph and check if the data is stationary using the ADF test.
4. Draw the ACF and PACF graphs to help choose the model settings.
5. Train ARIMA on the first 80% of the data and test it on the last 20%.
6. Predict the next 30 time periods (for example, the next 30 days).

Conclusion:
The ARIMA model learned the pattern of the traffic data and predicted future values. This can help in planning transport.
