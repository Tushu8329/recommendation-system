# Exp 1: Detect missing values, duplicate records & inconsistent entries (Healthcare)
import pandas as pd

# Dataset path - download ke baad yahan apna path likho
# (Mac: /Users/<name>/Downloads/file.csv | Windows: C:\Users\<name>\Downloads\file.csv)
DATA_PATH = r"/path/to/healthcare_dataset.csv"

df = pd.read_csv(DATA_PATH)
print("Shape:", df.shape)
print(df.head())
df.info()

# 1. Missing values
print("\nMissing values per column:\n", df.isnull().sum())
print("Total missing values:", df.isnull().sum().sum())

# 2. Duplicate records
print("\nDuplicate rows:", df.duplicated().sum())

# 3. Inconsistent entries
print("\nNames with irregular case (e.g. 'BoBBy jacKson'):", (df['Name'] != df['Name'].str.title()).sum())
print("Negative Billing Amount:", (df['Billing Amount'] < 0).sum())
print("Invalid Age (<0 or >120):", ((df['Age'] < 0) | (df['Age'] > 120)).sum())
adm = pd.to_datetime(df['Date of Admission'])
dis = pd.to_datetime(df['Discharge Date'])
print("Discharge date before admission:", (dis < adm).sum())
for col in ['Gender', 'Blood Type', 'Medical Condition', 'Admission Type', 'Test Results']:
    print(col, "->", df[col].unique().tolist())

# 4. Cleaning
df['Name'] = df['Name'].str.strip().str.title()          # fix inconsistent case
df = df.drop_duplicates()                                # remove duplicates
df = df[df['Billing Amount'] >= 0]                       # remove invalid billing
for col in df.columns:                                   # fill missing values
    if df[col].dtype.kind in 'if':
        df[col] = df[col].fillna(df[col].median())
    else:
        df[col] = df[col].fillna(df[col].mode()[0])
print("\nShape after cleaning:", df.shape)
print("Missing after cleaning:", df.isnull().sum().sum(), "| Duplicates:", df.duplicated().sum())

# Exp 2: Pearson correlation on PIMA Indians Diabetes dataset
import numpy as np
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
from scipy.stats import pearsonr

# Dataset path - download ke baad yahan apna path likho
# (Mac: /Users/<name>/Downloads/file.csv | Windows: C:\Users\<name>\Downloads\file.csv)
DATA_PATH = r"/path/to/diabetes.csv"

df = pd.read_csv(DATA_PATH)
print(df.head())

# Pearson formula: r = sum((x - x_mean)(y - y_mean)) / sqrt(sum((x - x_mean)^2) * sum((y - y_mean)^2))
def pearson(x, y):
    xd, yd = x - x.mean(), y - y.mean()
    return (xd * yd).sum() / np.sqrt((xd ** 2).sum() * (yd ** 2).sum())

print("\nGlucose vs Outcome (manual formula):", round(pearson(df['Glucose'], df['Outcome']), 4))
r, p = pearsonr(df['Glucose'], df['Outcome'])
print(f"Glucose vs Outcome (scipy): r = {r:.4f}, p-value = {p:.4e}")

# Full correlation matrix
corr = df.corr(method='pearson')
print("\nCorrelation of each feature with Outcome:\n", corr['Outcome'].sort_values(ascending=False))

plt.figure(figsize=(9, 7))
sns.heatmap(corr, annot=True, fmt='.2f', cmap='coolwarm')
plt.title('Pearson Correlation Matrix - PIMA Diabetes')
plt.tight_layout()
plt.show()

sns.regplot(x='Glucose', y='BMI', data=df, line_kws={'color': 'red'})
plt.title(f"Glucose vs BMI (r = {df['Glucose'].corr(df['BMI']):.2f})")
plt.show()

# Exp 3: Principal Component Analysis (PCA) on Mall Customer data
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA

# Dataset path - download ke baad yahan apna path likho
# (Mac: /Users/<name>/Downloads/file.csv | Windows: C:\Users\<name>\Downloads\file.csv)
DATA_PATH = r"/path/to/Mall_Customers.csv"

df = pd.read_csv(DATA_PATH)
print(df.head())

# Drop ID, encode Gender, scale features
X = pd.get_dummies(df.drop(columns='CustomerID'), drop_first=True).astype(float)
print("Features used:", list(X.columns))
X_scaled = StandardScaler().fit_transform(X)

# PCA with all components -> check explained variance
pca_full = PCA().fit(X_scaled)
print("\nExplained variance ratio:", pca_full.explained_variance_ratio_.round(3))
print("Cumulative variance     :", pca_full.explained_variance_ratio_.cumsum().round(3))

plt.plot(range(1, X.shape[1] + 1), pca_full.explained_variance_ratio_.cumsum(), marker='o')
plt.xlabel('Number of components'); plt.ylabel('Cumulative explained variance')
plt.title('Scree Plot'); plt.grid(); plt.show()

# Reduce to 2 components
pca = PCA(n_components=2)
X_pca = pca.fit_transform(X_scaled)
print("\nOriginal shape:", X.shape, "-> Reduced shape:", X_pca.shape)
print("Variance kept by 2 PCs:", round(pca.explained_variance_ratio_.sum() * 100, 2), "%")
print("\nPC loadings:\n", pd.DataFrame(pca.components_, columns=X.columns, index=['PC1', 'PC2']).round(3))

plt.scatter(X_pca[:, 0], X_pca[:, 1], c=df['Spending Score (1-100)'], cmap='viridis')
plt.colorbar(label='Spending Score')
plt.xlabel('PC1'); plt.ylabel('PC2'); plt.title('Customers in 2D (PCA)')
plt.show()

# Exp 7: End-to-end predictive analytics on finance data (build -> evaluate -> visualize)
import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder, StandardScaler
from sklearn.pipeline import make_pipeline
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score, classification_report, ConfusionMatrixDisplay

# Dataset path - download ke baad yahan apna path likho
# (Mac: /Users/<name>/Downloads/file.csv | Windows: C:\Users\<name>\Downloads\file.csv)
DATA_PATH = r"/path/to/Finance_data.csv"

df = pd.read_csv(DATA_PATH)
TARGET = 'Avenue'                       # target column (loan data ho to e.g. 'Loan_Status' / 'default')
if TARGET not in df.columns:
    TARGET = df.columns[-1]

# 1. Data understanding & cleaning
print(df.shape); print(df.head())
print("Missing:", df.isnull().sum().sum(), "| Duplicates:", df.duplicated().sum())
df = df.drop_duplicates().dropna(axis=1, how='all')
for col in df.columns:
    df[col] = df[col].fillna(df[col].mode()[0])

# 2. EDA
sns.countplot(y=df[TARGET].astype(str)); plt.title(f'Target: {TARGET}'); plt.tight_layout(); plt.show()

# 3. Encoding
for col in df.select_dtypes(exclude='number').columns:
    df[col] = LabelEncoder().fit_transform(df[col].astype(str))
X, y = df.drop(columns=TARGET), df[TARGET]

    model.fit(X_train, y_train)
    pred = model.predict(X_test)

# Exp 8: Twitter sentiment analysis (Positive / Negative / Neutral / Irrelevant)
import re
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import accuracy_score, classification_report, ConfusionMatrixDisplay

# Dataset path - download ke baad yahan apna path likho
# (Mac: /Users/<name>/Downloads/file.csv | Windows: C:\Users\<name>\Downloads\file.csv)
TRAIN_PATH = r"/path/to/twitter_training.csv"
TEST_PATH = r"/path/to/twitter_validation.csv"

cols = ['id', 'entity', 'sentiment', 'text']
train = pd.read_csv(TRAIN_PATH, header=None, names=cols).dropna()
test = pd.read_csv(TEST_PATH, header=None, names=cols).dropna()
print(train.head())

# Text cleaning
def clean(t):
    t = re.sub(r'http\S+|@\w+|#|[^a-zA-Z\s]', ' ', str(t).lower())
    return re.sub(r'\s+', ' ', t).strip()

train['clean'] = train['text'].apply(clean)
test['clean'] = test['text'].apply(clean)

train['sentiment'].value_counts().plot.bar(color=['tab:green', 'tab:red', 'tab:gray', 'tab:blue'])
plt.title('Sentiment Distribution'); plt.ylabel('Tweets'); plt.show()

# TF-IDF + Logistic Regression
vec = TfidfVectorizer(max_features=20000, ngram_range=(1, 2), stop_words='english')
X_train = vec.fit_transform(train['clean'])
X_test = vec.transform(test['clean'])

model = LogisticRegression(max_iter=1000)
model.fit(X_train, train['sentiment'])
pred = model.predict(X_test)

print("\nAccuracy:", round(accuracy_score(test['sentiment'], pred) * 100, 2), "%")
print(classification_report(test['sentiment'], pred))
ConfusionMatrixDisplay.from_predictions(test['sentiment'], pred, cmap='Blues', xticks_rotation=45)
plt.title('Confusion Matrix'); plt.tight_layout(); plt.show()

# Try your own tweets
for tweet in ["I love this game, it's amazing!", "Worst update ever, totally broken"]:
    print(tweet, "->", model.predict(vec.transform([clean(tweet)]))[0])

Time series analysis & forecasting with ARIMA

# Exp 9: Time series analysis & forecasting with ARIMA (transportation data)
import warnings
import pandas as pd
import matplotlib.pyplot as plt
from statsmodels.tsa.stattools import adfuller
from statsmodels.tsa.arima.model import ARIMA
from statsmodels.graphics.tsaplots import plot_acf, plot_pacf
from sklearn.metrics import mean_absolute_error, mean_squared_error
warnings.filterwarnings('ignore')

# Dataset path - download ke baad yahan apna path likho
# (Mac: /Users/<name>/Downloads/file.csv | Windows: C:\Users\<name>\Downloads\file.csv)
DATA_PATH = r"/path/to/transport.csv"

df = pd.read_csv(DATA_PATH)
print(df.head())
DATE_COL = None                         # e.g. 'Date'   (None = auto detect)
VALUE_COL = None                        # e.g. 'Vehicles' (None = auto detect)

if DATE_COL is None:
    DATE_COL = next(c for c in df.select_dtypes(exclude='number').columns
                    if pd.to_datetime(df[c], errors='coerce').notna().mean() > 0.8)
if VALUE_COL is None:
    nums = [c for c in df.select_dtypes('number').columns if 'id' not in c.lower()]
    VALUE_COL = max(nums, key=lambda c: df[c].nunique())
print("Date column:", DATE_COL, "| Value column:", VALUE_COL)

# Prepare series
df[DATE_COL] = pd.to_datetime(df[DATE_COL], errors='coerce')
ts = df.dropna(subset=[DATE_COL]).groupby(DATE_COL)[VALUE_COL].sum().sort_index()
step = ts.index.to_series().diff().median()
if step < pd.Timedelta('1D'):           # hourly/minute data -> daily total
    ts, step = ts.resample('D').sum(), pd.Timedelta('1D')
ts = ts.interpolate()

ts.plot(figsize=(12, 4), title=f'{VALUE_COL} over time'); plt.show()

# Stationarity check (ADF test)
p_value = adfuller(ts)[1]
print("ADF p-value:", round(p_value, 4), "->", "Stationary" if p_value < 0.05 else "Not stationary (use d=1)")
d = 0 if p_value < 0.05 else 1
lags = min(30, len(ts) // 2 - 1)
plot_acf(ts.diff().dropna() if d else ts, lags=lags); plot_pacf(ts.diff().dropna() if d else ts, lags=lags); plt.show()

# Train / test split + ARIMA
split = int(len(ts) * 0.8)
train, test = ts[:split], ts[split:]
model = ARIMA(train, order=(2, d, 2)).fit()
print(model.summary())
pred = model.forecast(len(test))
pred.index = test.index
print("MAE :", round(mean_absolute_error(test, pred), 3))
print("RMSE:", round(mean_squared_error(test, pred) ** 0.5, 3))

# Forecast next 30 periods
future = ARIMA(ts, order=(2, d, 2)).fit().forecast(30)
future.index = pd.date_range(ts.index[-1], periods=31, freq=pd.infer_freq(ts.index) or step)[1:]
print("\nNext 30 forecasts:\n", future.round(2))

plt.figure(figsize=(12, 5))
plt.plot(train, label='Train'); plt.plot(test, label='Test')
plt.plot(pred, label='ARIMA Prediction'); plt.plot(future, '--', label='Future Forecast')
plt.legend(); plt.title('ARIMA Forecast'); plt.show()

Missing values, duplicate records & inconsistent entries

Practical 1: Missing values, duplicate records & inconsistent entries

Problem Statement:
Download a real-world dataset (Healthcare / Finance) from an open data repository and detect missing values, duplicate records, and inconsistent entries.

Aim:
To find and fix missing values, duplicate records and inconsistent (wrong) entries in a real-world healthcare dataset.

Theory:
Real-world data is often messy. It can have empty cells, repeated rows and wrong values. Before we use the data for analysis, we must check it and clean it. This step is called data cleaning or data preprocessing.

The healthcare dataset has patient details like name, age, gender, blood type, medical condition, admission date, discharge date, billing amount and test results.

The main data quality problems are:
- Missing values: an empty cell in the data. It can happen when someone skips a field in a form, when a machine fails, or when two files are joined.
- Duplicate records: a row that appears more than once. Duplicates make counts wrong and give extra weight to the same record. Some duplicates only show up after cleaning, for example “Bobby” and “BOBBY” are the same name.
- Inconsistent entries: the same thing written in different ways, like “Male”, “male” and “M”, or dates in different formats.
- Invalid values: values that are not possible, like a negative age, a negative bill amount, or a discharge date before the admission date.
- Outliers: values that are very different from the others, like an age of 150. Outliers are not always wrong, so we check them before removing them.

How to handle missing values: we can remove the rows that have them (when there are very few), or we can fill them. Number columns are usually filled with the median (middle value), because the median is not affected by very large or very small values. Text columns are filled with the mode (most common value).

If the data is bad, the result will also be bad. This is called “garbage in, garbage out”. So data cleaning is always the first step of any analysis.

Formulas:
1. Missing % of a column = (number of empty cells ÷ total rows) × 100
2. Mean = (sum of all values) ÷ (number of values) = Σx ÷ n
3. Median = the middle value after sorting the data (if there are two middle values, take their average)
4. Mode = the value that appears the most times
5. Outlier rule (IQR method): IQR = Q3 − Q1. A value is an outlier if it is less than Q1 − 1.5 × IQR or more than Q3 + 1.5 × IQR
6. Example: salaries 20, 25, 30 and 1000 (in thousands). Mean = 1075 ÷ 4 = 268.75, but Median = (25 + 30) ÷ 2 = 27.5. The median gives the better picture.

Applications:
- Cleaning hospital and patient records
- Cleaning bank and customer data
- The first step of every data analysis and machine learning project

Procedure:
1. Load the healthcare dataset using pandas.
2. Look at the number of rows and columns, the first few rows and the data type of each column.
3. Count the empty cells in each column.
4. Count the repeated (duplicate) rows.
5. Find wrong entries: names in mixed case, negative bill amounts, impossible ages and discharge dates before admission dates.
6. Clean the data: fix the names, remove repeated rows and wrong rows, and fill the empty cells (median for numbers, mode for text).
7. Check the data again to make sure there are no empty cells or repeated rows left.

Conclusion:
We found the missing values, duplicate records and wrong entries in the healthcare dataset and fixed them. The data is now clean and ready for analysis.

Pearson correlation between variables

Practical 2: Pearson correlation between variables

Problem Statement:
Using a suitable real-world dataset (e.g., PIMA Indians Diabetes), implement Pearson correlation in Python to study relationships between variables.

Aim:
To find how the columns of the PIMA Indians Diabetes dataset are related to each other using Pearson correlation.

Theory:
Correlation tells us how two things change together. For example, when height goes up, weight usually goes up too.

Pearson correlation measures this relation as a number called r (the correlation coefficient). It works on number columns and checks for a straight-line (linear) relation.

The PIMA dataset has 768 patients. It has 8 medical values (pregnancies, glucose, blood pressure, skin thickness, insulin, BMI, diabetes pedigree function and age) and an Outcome column (1 = diabetes, 0 = no diabetes).

The value of r is always between −1 and +1:
- r = +1: perfect positive relation (both values go up together)
- r = −1: perfect negative relation (one value goes up, the other goes down)
- r = 0: no straight-line relation

Strength of the relation (ignore the + or − sign): 0 to 0.3 is weak, 0.3 to 0.7 is moderate, and 0.7 to 1 is strong.

p-value: it tells us if the relation is real or just by chance. If the p-value is less than 0.05, the relation is called statistically significant.

Correlation matrix: a table that shows r for every pair of columns. The diagonal is always 1, because every column is perfectly related to itself. When we colour this table, it is called a heatmap.

Important: correlation does not mean causation. Two things can change together without one causing the other. For example, ice-cream sales and swimming accidents both go up in summer, but ice-cream does not cause accidents.

Formulas:
1. Mean: x̄ = Σx ÷ n
2. Covariance: Cov(x, y) = Σ(x − x̄)(y − ȳ) ÷ (n − 1)
3. Standard deviation: s = √[Σ(x − x̄)² ÷ (n − 1)]
4. Pearson correlation: r = Cov(x, y) ÷ (sx × sy)
5. Same formula in one line: r = Σ(x − x̄)(y − ȳ) ÷ √[Σ(x − x̄)² × Σ(y − ȳ)²]
6. Significance test: t = r × √(n − 2) ÷ √(1 − r²)

Applications:
- Finding risk factors for diseases in medical research
- Finding which stocks move together in finance
- Choosing useful columns (feature selection) before building a model

Procedure:
1. Load the diabetes dataset using pandas and look at the first few rows.
2. Find r between Glucose and Outcome using the formula.
3. Check the answer using the pearsonr function, which also gives the p-value.
4. Find r for every pair of columns (the correlation matrix).
5. Sort the columns by their relation with Outcome to see which ones are most related to diabetes.
6. Draw a heatmap of the correlation matrix and a scatter plot with a line for two columns.

Conclusion:

Principal Component Analysis (PCA) to reduce features

Practical 3: Principal Component Analysis (PCA) to reduce features

Problem Statement:
Using the Mall customer segmentation data, apply Principal Component Analysis (PCA) to reduce the number of features.

Aim:
To reduce the number of columns (features) in the Mall Customers data using PCA.

Theory:
Dimensionality means the number of columns (features) in the data. When there are too many columns, the model becomes slow, the data is hard to show on a graph, and the model can learn noise. This problem is called the curse of dimensionality.

PCA (Principal Component Analysis) is a method to reduce the number of columns while keeping most of the information. It is an unsupervised method, which means it does not use any target (answer) column.

The Mall Customers dataset has 200 customers with these columns: CustomerID, gender, age, annual income and spending score.

PCA makes new columns called principal components (PCs). Each PC is a mix of the old columns.

- PC1 is the direction in which the data is spread out the most.
- PC2 is at a right angle to PC1 and keeps the next most spread, and so on.
- The PCs are not related to each other.

Variance means how spread out the data is. In PCA, more variance means more information.

Before PCA, we standardise the data, so that every column has mean 0 and standard deviation 1. If we do not, a column with big numbers (like income) will look more important than it really is.

Explained variance ratio tells us how much information each PC keeps. A scree plot shows this on a graph. We usually keep enough PCs to cover about 80% to 95% of the information.

Advantages: faster models, easy 2D or 3D graphs and less noise. Disadvantages: the new columns are hard to explain, and a little information is lost.

Formulas:
1. Standardisation: z = (x − mean) ÷ standard deviation
2. Covariance matrix: C = (Z^T × Z) ÷ (n − 1), where Z is the standardised data
3. Eigen equation: C × v = λ × v (v = eigenvector = direction of a PC, λ = eigenvalue = information in that PC)
4. Explained variance ratio of PC i = λi ÷ (λ1 + λ2 + … + λk)
5. New column: PC1 = a1·x1 + a2·x2 + … + ak·xk (the a values are called loadings)

Applications:
- Customer segmentation in marketing
- Image compression
- Face recognition
- Showing data with many columns on a 2D graph

Procedure:
1. Load the Mall Customers dataset.
2. Remove the CustomerID column and change Gender into numbers (0 and 1).
3. Standardise all columns so that they are on the same scale.
4. Apply PCA with all components and print the explained variance of each one.
5. Draw the scree plot (cumulative explained variance).
6. Apply PCA again with 2 components and print the loadings.
7. Draw a 2D scatter plot of the customers using PC1 and PC2.

Conclusion:
We reduced the data from 4 columns to 2 principal components. The explained variance shows how much information is kept, and the 2D graph makes the customers easy to see and compare.

Level: Moderate (all)

Random Forest Regression: a decision tree splits the data using simple yes/no questions, like “Is the height more than 5 m?”. Train and test split: we train the model on 80% of the data and test it on the other 20%, which the model has never seen. This shows how well the model works on new data.

6. MSE = Σ(y − ŷ)² ÷ n
7. RMSE = √MSE
8. R² = 1 − [Σ(y − ŷ)² ÷ Σ(y − ȳ)²]

SVM: is a tumour benign or malignant?

Practical 5: SVM: is a tumour benign or malignant?

Problem Statement:
Using the Cancer Dataset, implement a Support Vector Machine (SVM) model to predict whether a tumour is benign or malignant based on given diagnostic features.

Aim:
To predict whether a tumour is benign (not cancer) or malignant (cancer) using a Support Vector Machine (SVM).

Theory:
Classification means putting data into groups, called classes. Here there are two classes: benign (not cancer) and malignant (cancer). SVM (Support Vector Machine) is a supervised method used for classification.

The dataset has 569 patients (357 benign and 212 malignant). For each patient there are 30 values measured from images of cells, like radius, texture, perimeter, area and smoothness.

Main ideas of SVM:
- Hyperplane: the boundary that separates the two classes. With 2 columns it is a line, with 3 columns it is a flat plane, and with more columns it is called a hyperplane.
- Margin: many lines can separate the classes. SVM chooses the line with the widest gap between the two classes. This gap is called the margin. A wider margin usually gives fewer mistakes on new data.
- Support vectors: the points closest to the boundary. The boundary depends only on these points. That is why the method is called Support Vector Machine.
- Soft margin and C: real data is not perfectly separated, so SVM allows a few mistakes. A big C allows fewer mistakes but gives a narrow margin. A small C gives a wider margin but allows more mistakes.
- Kernel: if the classes cannot be separated by a straight line, SVM uses a kernel. A kernel moves the data to a higher dimension where a straight boundary can separate it. Common kernels are linear, polynomial and RBF. RBF is the most common.

Scaling: SVM works with distances, so all columns are brought to the same scale before training.

Confusion matrix: a table that shows right and wrong predictions.

- TP (True Positive): cancer, and the model said cancer
- TN (True Negative): not cancer, and the model said not cancer
- FP (False Positive): not cancer, but the model said cancer
- FN (False Negative): cancer, but the model missed it

In medical tests, FN is the most dangerous mistake, because a cancer patient is missed. So recall is a very important measure here.

Formulas:
1. Hyperplane: w·x + b = 0 (w = weights, b = bias)
2. Prediction rule: if w·x + b ≥ 0 then malignant, else benign
3. Margin width = 2 ÷ ||w|| (SVM makes this as large as possible)
4. Linear kernel: K(x, y) = x·y
5. RBF kernel: K(x, y) = exp(−γ × ||x − y||²)
6. Accuracy = (TP + TN) ÷ (TP + TN + FP + FN)
7. Precision = TP ÷ (TP + FP)
8. Recall = TP ÷ (TP + FN)
9. F1 score = 2 × Precision × Recall ÷ (Precision + Recall)

Applications:
- Cancer and disease detection
- Face detection
- Spam email filtering
- Handwriting recognition

Procedure:
1. Load the breast cancer dataset.
2. Remove the id column and the empty column.
3. Change the diagnosis column into numbers: M (malignant) = 1 and B (benign) = 0.
4. Split the data into training (80%) and testing (20%) parts, keeping the same ratio of classes in both.
5. Bring all columns to the same scale using StandardScaler.
6. Train the SVM model with the RBF kernel.
7. Predict on the test data and find accuracy, precision, recall and F1 score.
8. Draw the confusion matrix.

Conclusion:
The SVM model predicted whether a tumour is benign or malignant with high accuracy. This shows that SVM works well for medical diagnosis data.

Logistic Regression: is a customer likely to default?

Practical 6: Logistic Regression: is a customer likely to default?

Problem Statement:
Using a real-world finance dataset related to loan applicants. Build a Logistic Regression model to classify whether a customer is likely to default on a loan based on selected economic and demographic attributes.

Aim:
To predict whether a customer will default (not pay back) using a Logistic Regression model.

Theory:
Logistic Regression is a supervised method used for classification, mostly for yes/no questions. Here the question is: will the customer default or not?

Even though its name has “regression”, it is used for classification. It does not give a number like a price. It gives a probability between 0 and 1.

How it works:
- Step 1: it multiplies each input by a weight and adds them to get a score z.
- Step 2: it passes z through the sigmoid function, which changes any number into a value between 0 and 1. This value is the probability of default.
- Step 3: if the probability is 0.5 or more, the answer is “default” (1). Otherwise it is “no default” (0). This 0.5 is called the threshold, and it can be changed.

How it learns: the model changes its weights to make the log loss as small as possible. Log loss is high when the model is wrong and very sure about its answer.

Imbalanced data: the given dataset has 284,807 records, and only 492 of them are in class 1, which we treat as default. When one class is very small, the data is called imbalanced. A model can get 99.8% accuracy just by saying “no” every time, but it will miss every defaulter.

To handle this, we use class_weight = balanced, which gives more importance to the rare class. Other ways are oversampling (making more examples of the rare class, like SMOTE) and undersampling (using fewer examples of the big class).

So we do not trust accuracy alone. We also check precision, recall, F1 score and ROC-AUC. The ROC curve shows the true positive rate against the false positive rate at different thresholds. AUC is the area under this curve: 0.5 means random guessing and 1 means a perfect model.

Formulas:
1. Score: z = b0 + b1·x1 + b2·x2 + … + bn·xn
2. Sigmoid: p = σ(z) = 1 ÷ (1 + e^(−z))
3. Decision rule: if p ≥ 0.5 then default (1), else no default (0)
4. Odds = p ÷ (1 − p), and log(odds) = z
5. Log loss (for one record) = −[y × log(p) + (1 − y) × log(1 − p)]
6. Class weight (balanced) = total records ÷ (number of classes × records in that class)
7. Recall (True Positive Rate) = TP ÷ (TP + FN)
8. False Positive Rate = FP ÷ (FP + TN)
9. Precision = TP ÷ (TP + FP), F1 = 2 × Precision × Recall ÷ (Precision + Recall)

Applications:
- Loan approval and credit risk checking in banks
- Credit card fraud detection
- Disease prediction (yes/no)
- Spam email detection

Procedure:
1. Load the dataset and check how many records are in each class (Class 1 = default).
2. Take all columns except Class as inputs (X) and Class as the output (y).
3. Split the data into training (80%) and testing (20%) parts, keeping the same ratio of classes in both.
4. Bring all columns to the same scale using StandardScaler.
5. Train the Logistic Regression model with class_weight = balanced.
6. Predict on the test data and find the probability of default for each record.
7. Find accuracy, ROC-AUC, precision, recall and F1 score.
8. Draw the confusion matrix.

Conclusion:
The Logistic Regression model was able to find most of the defaulters. For imbalanced data, recall and ROC-AUC give a truer picture of the model than accuracy.

Level: Complex (any two)

Practical 8: Twitter sentiment analysis

Problem Statement:
Using real-world Twitter dataset, perform sentiment analysis to classify tweets based on their sentiment polarity.

Aim:
To find the feeling (sentiment polarity) of tweets: Positive, Negative, Neutral or Irrelevant.

Theory:
NLP (Natural Language Processing) helps computers understand human language. Sentiment analysis is an NLP task that finds the feeling in a piece of text.

Sentiment polarity means the type of feeling:
- Positive: “I love this game”
- Negative: “This update is so bad”
- Neutral: “The new patch comes out today”
- Irrelevant: tweets that are not about the topic (this extra class is in our dataset)

The dataset has about 74,000 training tweets and a separate validation file. Each tweet has an entity (the brand or game it talks about), a sentiment label and the tweet text.

There are three ways to do sentiment analysis:
- Rule-based (lexicon): each word has a score, like good = +1 and bad = −1, and the scores are added. Examples: TextBlob, VADER.
- Machine learning: a model learns from tweets whose feelings are already known. We use this way.
- Deep learning: models like LSTM and BERT. They are more accurate but need more computing power.

Text cleaning (preprocessing): make all letters small, remove links, @names, # symbols, numbers and punctuation, split the text into words (tokenization), and remove common words like “the”, “is” and “a” (stop words). Sometimes words are also cut to their root, like playing → play (stemming).

Text to numbers: a computer cannot read words, so we change the text into numbers. Bag of Words just counts each word. TF-IDF also counts words, but gives less value to words that are in almost every tweet and more value to words that are special to a tweet. We also use bigrams (two-word phrases like “not good”), so the model understands negative phrases.

Model: we train Logistic Regression on the TF-IDF numbers. Naive Bayes is another common choice. We check the model with accuracy, precision, recall and a confusion matrix.

Challenges: sarcasm (“Great, my phone died again”), slang, spelling mistakes, emojis and tweets with mixed feelings.

Formulas:
1. TF (term frequency) = (times the word appears in the tweet) ÷ (total words in the tweet)
2. IDF (inverse document frequency) = log(N ÷ number of tweets that have the word), where N = total tweets
3. TF-IDF = TF × IDF
4. Example (log base 10): N = 1000 tweets. “awesome” is in 10 tweets, so IDF = log(1000 ÷ 10) = log(100) = 2. “game” is in 500 tweets, so IDF = log(2) = 0.30. So “awesome” gets more value.
5. Softmax (for more than two classes): P(class k) = e^(zk) ÷ (sum of e^(z) for all classes)
6. Accuracy = correct predictions ÷ total predictions

Applications:
- Brand and product monitoring on social media
- Analysing product reviews
- Understanding customer feedback
- Studying public opinion and trends

Procedure:
1. Load the training and validation tweet files and give names to the columns.
2. Remove empty rows.
3. Clean the tweet text: small letters, remove links, @names, # and symbols.
4. Draw a bar graph of how many tweets have each sentiment.
5. Change the text into numbers using TF-IDF (single words and two-word phrases).
6. Train the Logistic Regression model on the training tweets.
7. Test the model on the validation tweets and find accuracy, precision and recall.
8. Draw the confusion matrix.
9. Try the model on some new tweets.

Conclusion:
We cleaned the tweets, changed them into TF-IDF numbers and trained a Logistic Regression model. The model was able to find the sentiment of most tweets correctly.

Time series analysis & forecasting with ARIMA

Practical 9: Time series analysis & forecasting with ARIMA

Problem Statement:
Using a suitable real-world time series dataset related to transportation traffic (such as road traffic volume, metro passenger count, railway ridership, or air traffic statistics), perform time series analysis (using ARIMA) and forecasting.

Aim:
To study transport traffic data over time and predict (forecast) future traffic using the ARIMA model.

Theory:
A time series is data recorded at regular times, like every hour, every day or every month. Examples are daily metro passengers, monthly air passengers and hourly road traffic. In a time series, the order of the data is very important, because past values affect future values.

Components of a time series:
- Trend: a long-term rise or fall (for example, passengers increase every year).
- Seasonality: a pattern that repeats at fixed times (more traffic on weekdays, more flights in December).
- Cyclic: long ups and downs without a fixed time (like economic cycles).
- Irregular (noise): random changes that cannot be predicted.

Stationarity: a time series is stationary when its average and spread do not change over time. ARIMA needs stationary data. We check this with the ADF (Augmented Dickey-Fuller) test. If the p-value is less than 0.05, the data is stationary. If not, we take the difference (subtract the previous value from each value) to remove the trend.

ACF and PACF: ACF shows how a value is related to its past values. PACF shows the direct relation with one past value, after removing the effect of the values in between. PACF helps us choose p, and ACF helps us choose q.

ARIMA(p, d, q) has three parts:
- AR (AutoRegressive), p: today's value depends on the last p values.
- I (Integrated), d: the number of times we take the difference to make the data stationary.
- MA (Moving Average), q: today's value depends on the last q errors.

For example, ARIMA(2, 1, 2) uses 2 past values, 1 difference and 2 past errors. To choose the best model, we try different values of p, d and q and pick the one with the lowest AIC.

Train and test: we never shuffle time series data. We train on the older 80% of the data and test on the newer 20%. Then we train on all the data and forecast the future.

If the data has strong seasonality, we can use SARIMA, which adds a seasonal part to ARIMA.

Formulas:
1. Time series: Y(t) = Trend + Seasonality + Noise
2. Differencing: y′(t) = y(t) − y(t−1)
3. AR(p): y(t) = c + φ1·y(t−1) + φ2·y(t−2) + … + φp·y(t−p) + e(t)
4. MA(q): y(t) = c + e(t) + θ1·e(t−1) + θ2·e(t−2) + … + θq·e(t−q)
5. AIC = 2k − 2·ln(L) (k = number of parameters, L = likelihood; lower AIC is better)
6. MAE = Σ|y − ŷ| ÷ n
7. RMSE = √[Σ(y − ŷ)² ÷ n]
8. MAPE = (100 ÷ n) × Σ|(y − ŷ) ÷ y| %

Applications:
- Planning metro, bus and train services
- Traffic control in cities
- Airline seat and flight planning
- Sales and weather forecasting

Procedure:
1. Load the transport dataset and find the date column and the value column.
2. Change the date column into proper dates and sort the data by date.
3. If the data is hourly, add it up to get daily totals. Fill any missing days.
4. Draw a graph of the data to see the trend and the pattern.
5. Check stationarity with the ADF test. If it is not stationary, use d = 1.
6. Draw the ACF and PACF graphs.
7. Split the data in time order: first 80% for training and last 20% for testing.
8. Train the ARIMA(2, d, 2) model and predict the test period.
9. Find MAE and RMSE to check the error.
10. Train the model on all the data, forecast the next 30 time periods and draw the graph.

Conclusion:
The ARIMA model learned the trend and pattern of the traffic data and forecast future values with a small error. Such forecasts help in planning transport services.
